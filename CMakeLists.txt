cmake_minimum_required(VERSION 3.16)
project(floating_point_hw2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

foreach(name q0_environment q1_benchmark q2_cmath q4_formatting q5_portability q6_extra)
    add_executable(${name} experiments/${name}/main.cpp)
endforeach()
target_sources(q1_benchmark PRIVATE experiments/q1_benchmark/kernels.cpp)
set_property(TARGET q1_benchmark PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
target_sources(q2_cmath PRIVATE experiments/q2_cmath/sqrt_paths.cpp)
target_sources(q5_portability PRIVATE experiments/q5_portability/operations.cpp)
target_sources(q6_extra PRIVATE experiments/q6_extra/operations.cpp)
add_executable(q2_libm experiments/q2_cmath/main.cpp experiments/q2_cmath/sqrt_paths.cpp)
target_compile_definitions(q2_libm PRIVATE HW2_FORCE_LIBM=1)

add_executable(q3_strict experiments/q3_fast_math/main.cpp experiments/q3_fast_math/operations.cpp)
add_executable(q3_fast experiments/q3_fast_math/main.cpp experiments/q3_fast_math/operations.cpp)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(q3_strict PRIVATE -O3 -fno-fast-math)
    target_compile_options(q3_fast PRIVATE -O3 -ffast-math)
    target_compile_options(q1_benchmark PRIVATE -O3 -fno-fast-math -ffp-contract=off)
    target_compile_options(q1_benchmark PRIVATE -fno-lto)
    target_link_options(q1_benchmark PRIVATE -fno-lto)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_compile_options(q1_benchmark PRIVATE -fno-tree-vectorize)
    else()
        target_compile_options(q1_benchmark PRIVATE -fno-vectorize -fno-slp-vectorize)
    endif()
    foreach(name q0_environment q1_benchmark q2_cmath q2_libm q3_strict q3_fast q4_formatting q5_portability q6_extra)
        set_property(TARGET ${name} PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)
        target_compile_options(${name} PRIVATE -Wall -Wextra -fno-lto -ffp-contract=off)
        target_link_options(${name} PRIVATE -fno-lto)
    endforeach()
    foreach(name q2_cmath q2_libm)
        target_compile_options(${name} PRIVATE -O2 -fno-fast-math -frounding-math)
    endforeach()
    foreach(name q4_formatting q5_portability q6_extra)
        target_compile_options(${name} PRIVATE -O2 -fno-fast-math -frounding-math)
    endforeach()
else()
    message(WARNING "The documented experiments require GCC-compatible floating-point flags; use WSL GCC.")
endif()
