cmake_minimum_required(VERSION 3.16)
project(floating_point_hw2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

foreach(name q0_environment q1_benchmark q2_cmath q4_formatting q5_portability q6_extra)
    add_executable(${name} experiments/${name}/main.cpp)
endforeach()
target_sources(q1_benchmark PRIVATE experiments/q1_benchmark/kernels.cpp)
set_property(TARGET q1_benchmark PROPERTY INTERPROCEDURAL_OPTIMIZATION FALSE)

add_executable(q3_strict experiments/q3_fast_math/main.cpp)
add_executable(q3_fast experiments/q3_fast_math/main.cpp)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(q3_strict PRIVATE -O3 -fno-fast-math)
    target_compile_options(q3_fast PRIVATE -O3 -ffast-math)
    target_compile_options(q1_benchmark PRIVATE -O3 -fno-fast-math -ffp-contract=off)
    target_compile_options(q1_benchmark PRIVATE -fno-lto)
    target_link_options(q1_benchmark PRIVATE -fno-lto)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_compile_options(q1_benchmark PRIVATE -fno-tree-vectorize)
    else()
        target_compile_options(q1_benchmark PRIVATE -fno-vectorize -fno-slp-vectorize)
    endif()
    target_compile_options(q2_cmath PRIVATE -fno-fast-math -fno-builtin-sin -fno-builtin-sinf -fno-builtin-sinl)
    foreach(name q4_formatting q5_portability q6_extra)
        target_compile_options(${name} PRIVATE -fno-fast-math -ffp-contract=off)
    endforeach()
else()
    message(WARNING "The documented experiments require GCC-compatible floating-point flags; use WSL GCC.")
endif()
